function createQueue(maxC,minI){
let running=0,lastStart=0,resolveIdle=null;
const q=[];
function next(){
if(running>=maxC)return;
if(!q.length){if(running===0&&resolveIdle){resolveIdle();resolveIdle=null}return}
const now=Date.now();
const wait=Math.max(0,lastStart+minI-now);
if(wait>0){setTimeout(next,wait);return}
const t=q.shift();
running++;
lastStart=Date.now();
t().finally(()=>{running--;setTimeout(next,0)});
}
return{
add(fn){return new Promise((res,rej)=>{q.push(()=>fn().then(res,rej));setTimeout(next,0)})},
idle(){return new Promise(r=>{if(running===0&&!q.length)return r();resolveIdle=r})}
};
}

async function buildBD() {
  if (S.selected.size < 5) { toast('Sélectionnez au moins 5 images','err'); return; }
  const selected = S.matrix.filter(m => S.selected.has(m.style + '_' + m.char));
  const refImages = selected.slice(0, 5);

  S.currentEpisode++;
  const seriesTitle = $('seriesTitle').value || 'Ma Série BD';
  const epTitle = seriesTitle + ' — Épisode ' + S.currentEpisode;

  log('════════════════════════', 'api');
  log('Épisode ' + S.currentEpisode + ' : ' + refImages.length + ' cases', 'ok');
  setStatus('Génération de l\'épisode ' + S.currentEpisode + '…');

  // 1. Utiliser les 5 images comme cases de BD
  const cases = refImages.map(m => ({ url: m.url, style: m.styleName, char: m.charName }));

  // 2. Générer un titre + dialogues pour la planche via l'IA texte
  let narrative = { title: epTitle, lines: [] };
  try {
    const prompt = 'Tu es scénariste de BD. À partir de ces ' + cases.length + ' images avec ces personnages : ' +
      cases.map(c => c.char).join(', ') +
      '. Écris un titre court et ' + cases.length + ' dialogues courts (max 12 mots chacun, un par case) pour raconter une mini-histoire. JSON : {"title":"...","lines":["...","...","...","...","..."]}';
    const raw = await callText([
      { role: 'system', content: 'Tu réponds UNIQUEMENT en JSON valide.' },
      { role: 'user', content: prompt }
    ], { temp: 0.9, max: 1000 });
    const match = raw.match(/\{[\s\S]*\}/);
    if (match) {
      const parsed = JSON.parse(match[0]);
      if (parsed.title) narrative.title = parsed.title;
      if (parsed.lines) narrative.lines = parsed.lines;
    }
  } catch (e) {
    log('Texte BD indisponible : ' + e.message, 'warn');
  }

  // 3. Ajouter à la série
  const episode = {
    num: S.currentEpisode,
    title: narrative.title,
    cases,
    lines: narrative.lines,
    ts: Date.now()
  };
  S.series.push(episode);

  renderEpisodes();
  renderHistory();
  document.getElementById('bdResultCard').style.display = 'block';
  document.getElementById('historyCard').style.display = 'block';
  S.bdImages = cases;
  setStatus('Épisode ' + S.currentEpisode + ' prêt');
  toast('Épisode ' + S.currentEpisode + ' créé','ok',3000);
  log('Épisode ' + S.currentEpisode + ' prêt : ' + narrative.title,'ok');
}

function renderEpisodes() {
  const box = document.getElementById('bdResult');
  box.innerHTML = S.series.map(ep => {
    let h = '<div class="ep"><strong>Épisode ' + ep.num + ' — ' + esc(ep.title) + '</strong>';
    h += '<div style="display:grid;grid-template-columns:repeat(5,1fr);gap:4px;margin-top:6px">';
    ep.cases.forEach((c, i) => {
      h += '<div style="text-align:center">';
      h += '<img src="' + c.url + '" style="width:100%;border-radius:4px;aspect-ratio:9/16;object-fit:cover">';
      if (ep.lines[i]) h += '<div style="font-size:0.62rem;color:var(--y);margin-top:3px;font-style:italic">« ' + esc(ep.lines[i]) + ' »</div>';
      h += '</div>';
    });
    h += '</div></div>';
    return h;
  }).join('');
}

function renderHistory() {
  const box = document.getElementById('historyList');
  box.innerHTML = S.series.map(ep =>
    '<div class="ep"><strong>Ép. ' + ep.num + '</strong> — ' + esc(ep.title) + ' · ' + ep.cases.length + ' cases · ' + new Date(ep.ts).toLocaleString('fr-FR') + '</div>'
  ).join('');
}

async function buildVideos() {
  if (!S.bdImages.length) { toast('Aucune image BD','err'); return; }
  if (S.running) return;
  S.running = true; S.stop = false;
  document.getElementById('genVideos').disabled = true;

  const episodes = S.series[S.series.length - 1];
  const cases = episodes.cases;

  log('════════════════════════', 'api');
  log('Génération de ' + cases.length + ' vidéos de 15 secondes', 'ok');
  setStatus('Création des vidéos (5 × 15s)…');

  const videos = [];

  for (let i = 0; i < cases.length - 1; i++) {
    if (S.stop) break;
    const first = cases[i].url;
    const last = cases[i + 1].url;
    const dialogue = episodes.lines[i] || '';
    const prompt = [
      'ONE SINGLE CONTINUOUS SHOT. NO CUTS.',
      'Smooth interpolation between the first and last keyframe.',
      'Style: ' + cases[i].style,
      'Character: ' + cases[i].char,
      dialogue ? 'Dialogue: ' + cases[i].char + ' says "' + dialogue + '"' : '',
      'Portrait 9:16 vertical, 24fps, no text, no watermarks.'
    ].filter(Boolean).join('. ');

    try {
      log('Vidéo ' + (i + 1) + '/' + (cases.length - 1) + ' — création tâche…','api');
      const videoId = await createVideo(first, last, prompt, 360);
      log('Vidéo ' + (i + 1) + ' — tâche : ' + videoId,'ok');
      setStatus('Vidéo ' + (i + 1) + ' — en cours…');

      const url = await pollVideo(videoId, (prog) => setStatus('Vidéo ' + (i + 1) + ' — ' + prog));
      videos.push({ url, scene: i + 1, dialogue });
      log('Vidéo ' + (i + 1) + ' terminée : ' + url.slice(0, 60) + '…','ok');
    } catch (e) {
      if (e.message === 'Arrêt demandé') break;
      log('Vidéo ' + (i + 1) + ' échec : ' + e.message,'err');
    }

    if (i < cases.length - 2 && !S.stop) {
      for (let r = 62; r > 0; r--) {
        if (S.stop) break;
        setStatus('Pause — prochaine vidéo dans ' + r + 's');
        await sleep(1000);
      }
    }
  }

  S.videos = videos;
  renderVideos();
  S.running = false;
  document.getElementById('genVideos').disabled = false;
  setStatus(videos.length + ' vidéos générées');
  toast(videos.length + ' vidéos prêtes','ok',3500);
}

function renderVideos() {
  const box = document.getElementById('bdResult');
  const vhtml = S.videos.map(v =>
    '<div style="margin-top:8px"><div style="font-size:0.7rem;color:var(--y);margin-bottom:4px">Vidéo scène ' + v.scene + (v.dialogue ? ' — « ' + esc(v.dialogue) + ' »' : '') + '</div>' +
    '<video src="' + v.url + '" controls playsinline style="width:100%;border-radius:6px;border:1px solid var(--border)"></video></div>'
  ).join('');
  box.innerHTML += '<div style="margin-top:12px;padding-top:8px;border-top:1px solid var(--border)">' + vhtml + '</div>';
}
